# ---------------- UTILIDADES ---------------- #

def limpar_moeda(valor):
    if valor in (None, ""):
        return 0.0
    if isinstance(valor, (int, float)):
        return float(valor)
    valor = str(valor).replace("R$", "").replace(" ", "").replace(".", "").replace(",", ".")
    try:
        return float(valor)
    except ValueError:
        return 0.0


def crc16(payload):
    crc = 0xFFFF
    for char in payload:
        crc ^= ord(char) << 8
        for _ in range(8):
            crc = ((crc << 1) ^ 0x1021) if crc & 0x8000 else crc << 1
            crc &= 0xFFFF
    return f"{crc:04X}"


def gerar_payload_pix(chave, valor, nome_favorecido):
    nome = nome_favorecido[:25].upper()
    valor_str = f"{valor:.2f}"

    payload = (
        "000201"
        "26360014BR.GOV.BCB.PIX01" + f"{len(chave):02}{chave}"
        "52040000"
        "5303986"
        f"54{len(valor_str):02}{valor_str}"
        "5802BR"
        f"59{len(nome):02}{nome}"
        "6008BRASILIA"
        "62070503***"
        "6304"
    )

    return payload + crc16(payload)

# ---------------- LOGIN ---------------- #

class LoginScreen:
    def __init__(self, root):
        self.root = root
        self.root.title("Sistema Financeiro da Marmoraria Jardim")
        self.root.geometry("400x500")

        if os.path.exists("img/logo_empresa.png"):
            img = Image.open("img/logo_empresa.png").resize((350, 180))
            self.logo = ImageTk.PhotoImage(img)
            tk.Label(root, image=self.logo).pack(pady=20)

        tk.Label(root, text="Usuário").pack()
        self.ent_user = tk.Entry(root, width=25)
        self.ent_user.pack(pady=5)

        tk.Label(root, text="Senha").pack()
        self.ent_pass = tk.Entry(root, show="*", width=25)
        self.ent_pass.pack(pady=5)

        ttk.Button(root, text="Entrar", command=self.autenticar).pack(pady=30)

    def autenticar(self):
        conn = get_conn()
        c = conn.cursor()
        c.execute(
            "SELECT perfil FROM usuarios WHERE login=%s AND senha=%s",
            (self.ent_user.get(), self.ent_pass.get())
        )
        res = c.fetchone()
        conn.close()

        if res:
            self.root.destroy()
            root = tk.Tk()
            AppFolha(root, res[0])
            root.mainloop()
        else:
            messagebox.showerror("Erro", "Login ou senha incorretos")

# ---------------- APP PRINCIPAL ---------------- #

class AppFolha:
    def __init__(self, root, perfil):
        self.root = root
        self.perfil = perfil
        self.root.title(f"Sistema Financeiro - {perfil}")
        self.root.geometry("1200x700")

        self.vars = {k: tk.StringVar() for k in [
            'nome', 'admissao', 'banco', 'pix',
            'salario', 'adiantamento', 'va'
        ]}

        self.funcionario_em_edicao = None

        self.criar_layout()
        self.atualizar_tabela()
        self.aplicar_permissoes()

    # -------- LAYOUT -------- #

    def criar_layout(self):
        frame_top = tk.Frame(self.root)
        frame_top.pack(fill="x", pady=5)

        tk.Label(frame_top, text=f"Perfil: {self.perfil}",
                 font=("Arial", 10, "bold")).pack(side="left", padx=10)

        tk.Button(frame_top, text="Log out", command=self.logout).pack(side="right", padx=10)

        self.notebook = ttk.Notebook(self.root)
        self.notebook.pack(fill="both", expand=True)

        self.aba_func = tk.Frame(self.notebook)
        self.notebook.add(self.aba_func, text="👥 Funcionários")

        self.montar_aba_funcionarios()

    def montar_aba_funcionarios(self):
        frame_form = tk.LabelFrame(self.aba_func, text="Cadastro / Edição", padx=10, pady=10)
        frame_form.pack(fill="x", padx=10, pady=10)

        campos = [
            ("Nome", "nome"),
            ("Admissão", "admissao"),
            ("Banco", "banco"),
            ("Chave PIX", "pix"),
            ("Salário R$", "salario"),
            ("Adiantamento R$", "adiantamento"),
            ("VA R$", "va")
        ]

        for i, (lbl, var) in enumerate(campos):
            tk.Label(frame_form, text=lbl).grid(row=0, column=i * 2, padx=5)
            tk.Entry(frame_form, textvariable=self.vars[var], width=15)\
                .grid(row=0, column=i * 2 + 1, padx=5)

        self.btn_salvar = tk.Button(
            frame_form, text="Salvar Funcionário",
            bg="#2ecc71", fg="white", command=self.salvar
        )
        self.btn_salvar.grid(row=1, column=0, columnspan=14, pady=10, sticky="nsew")

        cols = ("ID", "Nome", "Banco", "Salário", "Adiantamento", "VA", "Total")
        self.tree = ttk.Treeview(self.aba_func, columns=cols, show="headings")

        for c in cols:
            self.tree.heading(c, text=c)
            self.tree.column(c, anchor="center", width=140)

        self.tree.pack(fill="both", expand=True, padx=10, pady=5)

        frame_btn = tk.Frame(self.aba_func)
        frame_btn.pack(fill="x", padx=10, pady=10)

        tk.Button(frame_btn, text="Editar", command=self.editar).pack(side="left", padx=5)
        tk.Button(frame_btn, text="Excluir", command=self.excluir,
                  bg="#e74c3c", fg="white").pack(side="left", padx=5)
        tk.Button(frame_btn, text="Gerar PIX", command=self.gerar_pix,
                  bg="#3498db", fg="white").pack(side="left", padx=5)

    # -------- FUNÇÕES -------- #

    def atualizar_tabela(self):
        self.tree.delete(*self.tree.get_children())
        conn = get_conn()
        c = conn.cursor()
        c.execute("""
            SELECT id, nome, banco, salario_liquido, va, adiantamento
            FROM funcionarios
            ORDER BY nome
        """)
        for r in c.fetchall():
            total = float(r[3]) + float(r[4])
            self.tree.insert("", "end", values=(
                r[0], r[1], r[2],
                f"{r[3]:.2f}", f"{r[4]:.2f}", f"{r[5]:.2f}",
                f"R$ {total:.2f}"
            ))
        conn.close()

    def salvar(self):
        f = self.funcionario_em_edicao or Funcionario()

        f.nome = self.vars['nome'].get()
        f.admissao = self.vars['admissao'].get()
        f.banco = self.vars['banco'].get()
        f.chave_pix = self.vars['Chave PIX'].get()
        f.salario = limpar_moeda(self.vars['salario R$'].get())
        f.adiantamento = limpar_moeda(self.vars['adiantamento R$'].get())
        f.va = limpar_moeda(self.vars['va R$'].get())

        f.salvar()

        self.funcionario_em_edicao = None
        self.btn_salvar.config(text="Salvar Funcionário")
        for v in self.vars.values():
            v.set("")

        self.atualizar_tabela()

    def editar(self):
        sel = self.tree.selection()
        if not sel:
            return
        fid = self.tree.item(sel)['values'][0]
        f = Funcionario.buscar_por_id(fid)
        if not f:
            return

        self.funcionario_em_edicao = f
        self.vars['nome'].set(f.nome)
        self.vars['admissao'].set(f.admissao)
        self.vars['banco'].set(f.banco)
        self.vars['Chave PIX'].set(f.chave_pix)
        self.vars['salario R$'].set(f"{f.salario:.2f}")
        self.vars['adiantamento R$'].set(f"{f.adiantamento:.2f}")
        self.vars['va R$'].set(f"{f.va:.2f}")

        self.btn_salvar.config(text="Confirmar Alteração")

    def excluir(self):
        if self.perfil != "Master":
            return
        sel = self.tree.selection()
        if not sel:
            return
        fid = self.tree.item(sel)['values'][0]

        if messagebox.askyesno("Excluir", "Deseja remover este funcionário?"):
            conn = get_conn()
            c = conn.cursor()
            c.execute("DELETE FROM funcionarios WHERE id=%s", (fid,))
            conn.commit()
            conn.close()
            self.atualizar_tabela()

    def gerar_pix(self):
        sel = self.tree.selection()
        if not sel:
            return
        it = self.tree.item(sel)['values']
        valor = float(it[3].replace(",", "."))
        payload = gerar_payload_pix("PIX_CHAVE", valor, it[1])

        win = tk.Toplevel(self.root)
        win.title("QR Code PIX")

        img = qrcode.make(payload).resize((250, 250))
        im = ImageTk.PhotoImage(img)
        tk.Label(win, image=im).pack(pady=10)
        win.image = im

    def aplicar_permissoes(self):
        if self.perfil == "Financeiro":
            self.btn_salvar.config(state="disabled")

    def logout(self):
        self.root.destroy()
        root = tk.Tk()
        LoginScreen(root)
        root.mainloop()

# ---------------- START ---------------- #

if __name__ == "__main__":
    root = tk.Tk()
    LoginScreen(root)
    root.mainloop()
